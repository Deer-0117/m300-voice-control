cmake_minimum_required(VERSION 3.16)
project(m300_voice_psdk_host C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(M300_GROUND_ONLY "Force no joystick acquisition or movement in this build" ON)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  message(FATAL_ERROR "Build on the Jetson ARM64 device")
endif()
foreach(required DJI_PSDK_ROOT M300_PACKAGE_ROOT M300_HOST_SUPPORT)
  if(NOT DEFINED ${required})
    message(FATAL_ERROR "Missing ${required}")
  endif()
endforeach()
get_filename_component(DJI_PSDK_ROOT "${DJI_PSDK_ROOT}" REALPATH)
get_filename_component(M300_PACKAGE_ROOT "${M300_PACKAGE_ROOT}" REALPATH)
add_compile_definitions(_GNU_SOURCE SYSTEM_ARCH_LINUX PLATFORM_ARCH_aarch64=1)
include_directories(
  ../../../module_sample ../../../../sample_c/module_sample ../common
  "${CMAKE_CURRENT_SOURCE_DIR}/application" "${DJI_PSDK_ROOT}/psdk_lib/include")
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../common/3rdparty")
set(ORIGINAL_COMMON "${DJI_PSDK_ROOT}/samples/sample_c++/platform/linux/common")
set(ORIGINAL_HAL "${DJI_PSDK_ROOT}/samples/sample_c++/platform/linux/manifold2/hal")
file(GLOB_RECURSE MODULE_COMMON_SRC "${ORIGINAL_COMMON}/*.c" "${ORIGINAL_COMMON}/*.cc" "${ORIGINAL_COMMON}/*.cpp")
file(GLOB_RECURSE MODULE_HAL_SRC "${ORIGINAL_HAL}/*.c" "${ORIGINAL_HAL}/*.cc" "${ORIGINAL_HAL}/*.cpp")
file(GLOB_RECURSE MODULE_APP_SRC application/*.c application/*.cc application/*.cpp)
list(FILTER MODULE_APP_SRC EXCLUDE REGEX "/main\\.cpp$")
set(UTIL_ROOT "${DJI_PSDK_ROOT}/samples/sample_c/module_sample/utils")
set(MODULE_UTIL_SRC
  "${UTIL_ROOT}/cJSON.c" "${UTIL_ROOT}/dji_config_manager.c"
  "${UTIL_ROOT}/util_file.c" "${UTIL_ROOT}/util_md5.c" "${UTIL_ROOT}/util_misc.c")
add_subdirectory("${M300_PACKAGE_ROOT}/psdk" "${CMAKE_BINARY_DIR}/m300-runtime")
add_executable(${PROJECT_NAME} ${MODULE_APP_SRC} ${MODULE_COMMON_SRC} ${MODULE_HAL_SRC}
  ${MODULE_UTIL_SRC} "${M300_HOST_SUPPORT}/migration_main.cpp")
target_compile_options(${PROJECT_NAME} PRIVATE -pthread)
if(M300_GROUND_ONLY)
  target_compile_definitions(${PROJECT_NAME} PRIVATE M300_GROUND_ONLY=1)
else()
  # Flight acceptance is a separate phase; do not silently build an enabled host.
  message(FATAL_ERROR "This host add-on is ground-validation only. Flight build is not released yet.")
endif()
find_package(OpenCV QUIET)
find_package(FFMPEG REQUIRED)
find_package(OPUS REQUIRED)
find_package(LIBUSB REQUIRED)
target_include_directories(${PROJECT_NAME} PRIVATE "${UTIL_ROOT}"
  ${OpenCV_INCLUDE_DIRS} ${FFMPEG_INCLUDE_DIR} ${OPUS_INCLUDE_DIR} ${LIBUSB_INCLUDE_DIR})
target_compile_definitions(${PROJECT_NAME} PRIVATE FFMPEG_INSTALLED OPUS_INSTALLED LIBUSB_INSTALLED)
if(OpenCV_FOUND)
  target_compile_definitions(${PROJECT_NAME} PRIVATE OPEN_CV_INSTALLED)
endif()
set(PAYLOAD_LIBRARY "${DJI_PSDK_ROOT}/psdk_lib/lib/aarch64-linux-gnu-gcc/libpayloadsdk.a")
if(NOT EXISTS "${PAYLOAD_LIBRARY}")
  message(FATAL_ERROR "Device ARM64 libpayloadsdk.a not found")
endif()
# Static runtime first, DJI archive second, then platform dependency libraries.
target_link_libraries(${PROJECT_NAME} PRIVATE m300_voice_runtime "${PAYLOAD_LIBRARY}"
  ${FFMPEG_LIBRARIES} ${OPUS_LIBRARY} ${LIBUSB_LIBRARY} usb-1.0 ${OpenCV_LIBS} pthread m)
set_target_properties(${PROJECT_NAME} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
